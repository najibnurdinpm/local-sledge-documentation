# Product Review

Make the appearance of the Sledge Product Review app remain in line with the theme you have using the custom components that Sledge provides.

Available custom components :

- [Review Product Info](#review-product-info)

## Review Product Info

Review Product Info is custom component, will change product information each review cards.<br/><br/>

            ```tsx filename="app/components/SledgeReviewProductInfo.tsx" showLineNumbers copy
            import { Link } from '@remix-run/react';

            interface IReviewProductInfo {
                product: {name: string; url: string};
            }

            export function SledgeReviewProductInfo({product}: IReviewProductInfo) {
                const {name, url = '#'} = product || {};

                return (
                    <div className="sledge-product-review__widget-about-product-text">
                        About{' '}
                        <Link
                            className="sledge-product-review__widget-about-product-link"
                            to={'/products/' + url}
                            onClick={(e) => e.stopPropagation()}
                        >
                            {name}
                        </Link>
                    </div>
                );
            }
            ```

            <h3 className="nx-font-semibold nx-tracking-tight nx-text-slate-900 dark:nx-text-slate-100 nx-text-2xl">Call custom component to prop</h3>

            > <div className="not-italic">`<CustomComponent>` must be added as a child of another sledge's widget.</div>

            ```tsx {8} showLineNumbers copy
            import { CustomComponents } from "@sledge-app/core";
            import { HappyCustomersPage } from '@sledge-app/react-product-review';
            import SledgeHappyCustomersPage from "~/components/SledgeHappyCustomersPage";

            export default function SledgeHappyCustomersPage() {
                return (
                <HappyCustomersPage.Root>
                    <CustomComponents
                        reviewProductInfo={({product}: any) => {
                            return <SledgeReviewProductInfo product={product} />;
                        }}
                    />
                    <HappyCustomersPage.Header />
                    <HappyCustomersPage.List />
                </HappyCustomersPage.Root>
                );
            }
            ```

            ```html showLineNumbers copy
            <script>
                function renderReviewProductInfo({ product }) {
                    const { name, url = '#' } = product || {};

                    return `
                        <div class="sledge-product-review__widget-about-product-text">
                        About 
                            <a href="/products/${url}" class="sledge-product-review__widget-about-product-link" onclick="handleClickReviewProductLink(event)">
                                ${name}
                            </a>
                        </div>
                    `;
                }
            </script>
            ```

            <h3 className="nx-font-semibold nx-tracking-tight nx-text-slate-900 dark:nx-text-slate-100 nx-text-2xl">Call custom component to attribute</h3>

            Only provide the function name (without `()`), see the [API reference](/html-data-attribute#data-render-review-product-info)

            > <div className="not-italic">`<div data-component="custom-components">` must be added as a child of another sledge's widget.</div>

            ```html {6} showLineNumbers copy
            <div

                data-width-element="1170px"

            >
                <div 
                    data-component="custom-components"
            ```
